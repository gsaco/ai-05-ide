\begin{frame}{Two dimensions govern the distribution of gains}
\begin{columns}[T]
\begin{column}{.47\textwidth}
\begin{block}{Autonomy: what roles can AI take?}
\textbf{Autonomous:} independent producer, worker, or solver.\\[3mm]
\textbf{Non-autonomous:} solver / co-pilot only.
\end{block}
\end{column}
\begin{column}{.47\textwidth}
\begin{block}{Capability: what can AI solve?}
AI knows solutions up to difficulty $a=z_{AI}<1$.\\[3mm]
A permitted role need not be profitable at a low capability.
\end{block}
\end{column}
\end{columns}
\take{The question is who earns more \emph{in equilibrium}, after occupations, matches, wages, and compute rents adjust.}
\src{Road map: propositions $\to$ three-type derivation $\to$ computation $\to$ Lean boundary.}
\end{frame}

\begin{frame}{The agent's problem: allocate time and knowledge}
A worker knows $z$; a solver knows $s>z$; difficulty $x\sim U[0,1]$.
\[
\underbrace{h\,n(z)(1-z)}_{\text{solver time used}}=1
\quad\Longrightarrow\quad n(z)=\frac{1}{h(1-z)}.
\]
\[
\max_{\text{organization},\,z,s}\ \Pi,
\qquad \Pi_{HH}=n(z)[s-w(z)]-w(s).
\]
\begin{itemize}\setlength\itemsep{3mm}
\item Humans choose income-maximizing roles; compute owners seek rent $r$.
\item Competitive entry: active firms earn zero profit; all resource markets clear.
\item Output per worker is $s$, including problems solved without assistance.
\end{itemize}
\src{Section 3.1 in both versions. AI replaces the relevant wage by its compute rental rate.}
\end{frame}

\begin{frame}{The headline results have important conditions}
\begin{itemize}\setlength\itemsep{3mm}
\item Unit mass of humans, one unit of time each; observable $z\in[0,1]$ with continuous, strictly positive density $g$.
\item Uniform independent difficulties, risk neutrality, output price one; competitive firms, no fixed costs, at most two layers.
\item $0<h<h_0$: no independent human producers before AI.
\item $a\in[0,1)$; opportunities exceed human-plus-AI production capacity.
\item Compute abundant relative to human time. For Proposition 6, compare the \emph{same} capability and compute supply.
\end{itemize}
\take{The top-winner claim needs both $h<h_0$ and $a<1$. Perfect AI is a different boundary case.}
\src{Sections 3.1, 3.3, 5.2, and 6. Exact sufficient compute bound in the appendix slide.}
\end{frame}

\begin{frame}{Proposition 5: bottom winners require capability}
Define autonomous-AI winners relative to no AI:
\[
B=\{z\in[0,a]:\wa(z)>w(z)\},\qquad
T=\{z\in[a,1]:\wa(z)>w(z)\}.
\]
\begin{block}{Under the stated model conditions}
\[
B\ne\varnothing\ \Longleftrightarrow\ a>\bar a,
\qquad \bar a\in\operatorname{int}W;
\qquad T\ne\varnothing\quad\forall a\in[0,1).
\]
\end{block}
$W$ is the \textbf{pre-AI worker set}. Winners lie at the extremes; humans with knowledge $a$ lose.
\take{Even ``basic'' AI can help the bottom: $\bar a$ lies \emph{inside} the worker region. Autonomy alone does not settle the sign.}
\src{Proposition 5: v11 p.24; course PDF p.25. $\wa$ denotes the paper's $w^*$.}
\end{frame}

\begin{frame}{Proposition 6: free co-pilots still need to be useful}
\[
r^N=0,\qquad
\begin{cases}
a\le w(0):&\text{AI unused; pre-AI wages and occupations unchanged},\\
a>w(0):&\text{lowest-knowledge humans use AI advice}.
\end{cases}
\]
\begin{itemize}\setlength\itemsep{3mm}
\item Near the bottom: $\wn(z)\ge\max\{w(z),\wa(z)\}$, \textbf{strict} if $a>w(0)$.
\item Some $z>0$ has $\wn(z)\le w(z)$, \textbf{strict} if $a>w(0)$.
\item Near the top: $\wn(z)\le\wa(z)$, strict away from $z=1$.
\item Output: $Y^A>Y^N$. Efficiency is relative to the permitted technology.
\end{itemize}
\take{The thresholds $w(0)$ and $\bar a$ answer different questions. Neither can be dropped.}
\src{Proposition 6: v11 p.27; course PDF p.28. Full quantifiers and occupational clauses in the appendix.}
\end{frame}

\begin{frame}{Why the bottom and top respond differently}
\begin{columns}[T]
\begin{column}{.48\textwidth}
\textbf{Bottom: matching versus income share}
\[
w(z)=s-h(1-z)w(s).
\]
A worse solver reduces output; a cheaper solver lets the worker retain more. Capability changes the balance.
\end{column}
\begin{column}{.48\textwidth}
\textbf{Top: scale up scarce expertise}
\[
\wa(s)=\frac{s-a}{h(1-a)}
\]
when supervising AI workers. Low-cost routine work increases the return to human problem-solving.
\end{column}
\end{columns}
\vspace{5mm}
Autonomous compute has an outside option: $r^A=a$. Co-pilot-only compute can remain idle: $r^N=0$.
\src{Wage formulas apply to their respective active roles; they are not global wage formulas.}
\end{frame}

\begin{frame}{A three-type economy we can solve by hand}
\[
(z_L,z_M,z_H)=(0,\tfrac12,1),\quad
(m_L,m_M,m_H)=(\tfrac{11}{20},\tfrac3{10},\tfrac3{20}),\quad
h=\tfrac12,\quad\mu=2.
\]
Let wages be $(u,v,t)$. Three active human teams imply
\[
u+\tfrac12v=\tfrac12,\qquad u+\tfrac12t=1,\qquad v+\tfrac14t=1.
\]
\[
t-v=1\quad\Rightarrow\quad
\boxed{(u,v,t)=(\tfrac15,\tfrac35,\tfrac85)}.
\]
Worker masses $(L\!\to\!M,L\!\to\!H,M\!\to\!H)=(.32,.23,.14)$ clear all three markets; $Y_0=.53$.
\take{This is a finite-type analogue. It changes the source's density assumption and does not inherit its continuum theorems.}
\end{frame}

\begin{frame}{Autonomous AI: derive an equilibrium branch}
Take $0<a<\tfrac13$. Low humans work with middle solvers; the remaining middle humans and all high humans supervise AI.
\[
r^A=a,\qquad
t=\frac{1-a}{(1-a)/2}=2,\qquad
v=\frac{\frac12-a}{(1-a)/2}=\frac{1-2a}{1-a}.
\]
\[
u=\tfrac12-\tfrac12v=\boxed{\frac{a}{2(1-a)}}.
\]
\begin{itemize}\setlength\itemsep{2mm}
\item Low workers use $.55/2=.275$ middle solvers, leaving $.025$.
\item AI workers use $7/[20(1-a)]<2$ compute; the rest produces independently.
\item The other no-entry inequalities hold; $v>\tfrac12$ on this branch.
\end{itemize}
\src{An allocation plus a feasible no-profit price vector is the equilibrium certificate.}
\end{frame}

\begin{frame}{The threshold is visible in three lines of algebra}
\[
\wa_L>w_L
\quad\Longleftrightarrow\quad
\frac{a}{2(1-a)}>\frac15
\]
\[
\Longleftrightarrow\quad 5a>2(1-a)
\quad\Longleftrightarrow\quad\boxed{a>\frac27}.
\]
Multiplication preserves the inequality because $1-a>0$.
\begin{center}
\begin{tabular}{lcc}
\toprule
Same autonomy; different capability & $a=1/5$ & $a=3/10$\\
\midrule
Lowest wage & $1/8<1/5$ & $3/14>1/5$\\
Verdict & loses & gains\\
\bottomrule
\end{tabular}
\end{center}
\src{Both values lie in $0<a<1/3$. Equality at $a=2/7$ is not a strict gain. This is not the continuum's numerical threshold.}
\end{frame}

\begin{frame}{Co-pilot-only AI: solve the same economy at $a=.4$}
Spare compute is idle, so $r^N=0$. Low humans using AI earn $u=a=.4$.
\[
u+\tfrac12t=1\ \Rightarrow\ t=1.2,
\qquad v+\tfrac14t=1\ \Rightarrow\ v=.7.
\]
\begin{columns}[T]
\begin{column}{.48\textwidth}
\textbf{Allocation}
\begin{itemize}
\item $.30$ middle workers use high solvers.
\item $.15$ low workers use high solvers.
\item $.40$ low workers use AI advice.
\end{itemize}
\end{column}
\begin{column}{.48\textwidth}
\textbf{Income comparison}
\[
(.2,.6,1.6)\longrightarrow(.4,.7,1.2).
\]
Bottom and middle gain; top loses.\\[2mm]
$Y^N=.61$; all of it is labor income.
\end{column}
\end{columns}
\src{At $a=.2$, the reported non-autonomous equilibrium remains the pre-AI one: the capability boundary matters.}
\end{frame}

\begin{frame}{What the checks teach us}
\begin{enumerate}\setlength\itemsep{4mm}
\item \textbf{Bottom:} capability crosses a threshold, even with autonomy fixed.
\item \textbf{Top:} autonomous AI scales scarce expertise; the source result needs $h<h_0$ and $a<1$.
\item \textbf{Output:} more feasible uses of compute raise output, without making every human better off.
\item \textbf{Verification:} an algebraic proof, an equilibrium computation, and a continuum theorem are different achievements.
\end{enumerate}
\take{The defensible conclusion is a two-dimensional taxonomy, with domains and inequalities stated explicitly.}
\end{frame}

\begin{frame}{Reference: complete Proposition 6 clauses}
\small
The non-autonomous equilibrium is unique, efficient, maximizes labor income within that technology, and $r^N=0$.
If $a\le w(0)$, AI is unused. If $a>w(0)$,
\[
W_a^N\preceq W_p^N\cup I^N\cup S_p^N;
\qquad W_a^N,W_p^N,S_p^N\ne\varnothing
\]
and only $I^N$ may be empty. In either case:
\begin{align*}
&Y^A>Y^N;\\
&\exists z\in(0,1]:\ \wn(z)\le w(z)\quad(\text{strict if }a>w(0));\\
&\exists\epsilon>0\ \forall z\in[0,\epsilon):\ \wn(z)\ge\max\{w(z),\wa(z)\}\\
&\hspace{52mm}(\text{strict if }a>w(0));\\
&\exists\epsilon>0\ \forall z\in(1-\epsilon,1]:\ \wn(z)\le\wa(z)\quad(\text{strict if }z\ne1).
\end{align*}
\src{Source: v11 p.27 / course p.28. $W_a$: AI-assisted; $W_p$: human-assisted; $I$: independent; $S_p$: human solvers.}
\end{frame}
